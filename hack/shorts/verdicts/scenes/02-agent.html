review.yml

agents:
- name: reviewer
  source: { model: openrouter/qwen/qwen3.7-flash }
  tools: [read_file, search_files, write_file]
  # no edit_file: it reviews, it cannot change code

jobs:
- name: review
  plan:
  - get: repo
  - agent: reviewer
    inputs: [repo]
    outputs: [report]
    messages:
      - Write a risk summary to report/summary.md.
    verdicts:
      - approve: publish   # the decision picks the step
      - reject: escalate
    assert:
      files: [report/summary.md]  # must exist
  - task: escalate
    run: echo paging a human
  - task: publish
    inputs: [report]
    run: cat report/summary.md
